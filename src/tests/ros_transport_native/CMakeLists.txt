cmake_minimum_required(VERSION 3.16)
project(drone_ros_transport_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
if(NOT DRONE_REPO_ROOT)
    message(FATAL_ERROR "DRONE_REPO_ROOT is required")
endif()
add_executable(ros_transport_native_tests
    ${DRONE_REPO_ROOT}/src/tests/ros_transport_test.cpp
    ${DRONE_REPO_ROOT}/src/platform/cpu_affinity.cpp)
target_include_directories(ros_transport_native_tests PRIVATE ${DRONE_REPO_ROOT}/src)
target_compile_definitions(ros_transport_native_tests PRIVATE DRONE_TRACKER_RK3576)
target_link_libraries(ros_transport_native_tests yaml-cpp Threads::Threads)
ament_target_dependencies(ros_transport_native_tests rclcpp std_msgs)
