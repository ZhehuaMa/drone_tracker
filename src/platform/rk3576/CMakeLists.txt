cmake_minimum_required(VERSION 3.16)
project(rk3576_yolo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(RKNN_INCLUDE_DIR "" CACHE PATH "RKNN SDK include directory (target Runtime 2.3.2; existing build SDK 2.3.0)")
set(RKNN_LIBRARY "" CACHE FILEPATH "ARM64 librknnrt.so for linking (target Runtime 2.3.2; existing build SDK 2.3.0)")
if(NOT EXISTS "${RKNN_INCLUDE_DIR}/rknn_api.h" OR NOT EXISTS "${RKNN_LIBRARY}")
    message(FATAL_ERROR "RK3576 standalone backends target RKNN Runtime 2.3.2 and driver 0.9.8. Set RKNN_INCLUDE_DIR to the SDK directory containing rknn_api.h and RKNN_LIBRARY to ARM64 librknnrt.so. Existing validation used build SDK 2.3.0 with target Runtime 2.3.2; rebuilding that SDK image is not required.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    message(FATAL_ERROR "RK3576 standalone entry requires an ARM64 target environment")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn video imgcodecs videoio)
set(DRONE_TRACKER_COMMON_STANDALONE ON)
option(DRONE_TRACKER_BUILD_TESTS "Build common and RKNN fault injection tests" ON)
enable_testing()
add_subdirectory(../../hybrid_tracker common)
add_library(rk3576_yolo_host STATIC yolo_preprocessor.cpp nanotrack_preprocessor.cpp host_frame.cpp)
target_link_libraries(rk3576_yolo_host PUBLIC hybrid_tracker)
add_library(rk3576_yolo_runtime STATIC rknn_session.cpp rknn_runtime.cpp rknn_yolo_contract.cpp rknn_yolo_backend.cpp
    rknn_nanotrack_contract.cpp rknn_nanotrack_backend.cpp)
target_include_directories(rk3576_yolo_runtime PRIVATE "${RKNN_INCLUDE_DIR}")
target_link_libraries(rk3576_yolo_runtime PUBLIC rk3576_yolo_host)
add_library(rk3576_yolo_sdk STATIC rknn_api.cpp rknn_yolo_factory.cpp rknn_nanotrack_factory.cpp)
target_include_directories(rk3576_yolo_sdk PRIVATE "${RKNN_INCLUDE_DIR}")
target_link_libraries(rk3576_yolo_sdk PUBLIC rk3576_yolo_runtime "${RKNN_LIBRARY}")
add_executable(rk3576_yolo_probe rk3576_yolo_probe.cpp)
target_include_directories(rk3576_yolo_probe PRIVATE "${RKNN_INCLUDE_DIR}")
target_link_libraries(rk3576_yolo_probe PRIVATE rk3576_yolo_sdk)
add_executable(rk3576_nanotrack_probe rk3576_nanotrack_probe.cpp)
target_link_libraries(rk3576_nanotrack_probe PRIVATE rk3576_yolo_sdk)
if(DRONE_TRACKER_BUILD_TESTS)
    foreach(test_name rknn_session rknn_nanotrack_backend nanotrack_preprocess)
        add_executable(${test_name}_tests tests/${test_name}_test.cpp)
        target_include_directories(${test_name}_tests PRIVATE "${RKNN_INCLUDE_DIR}")
        target_link_libraries(${test_name}_tests PRIVATE rk3576_yolo_runtime)
        add_test(NAME ${test_name}_tests COMMAND ${test_name}_tests)
    endforeach()
    add_test(NAME nanotrack_probe_missing_args COMMAND rk3576_nanotrack_probe --mode replay)
    add_test(NAME nanotrack_probe_invalid_bbox COMMAND rk3576_nanotrack_probe --template-model unused
        --search-model unused --head-model unused --mode replay --video unused --bbox 0,0,-1,1)
    set_tests_properties(nanotrack_probe_missing_args nanotrack_probe_invalid_bbox PROPERTIES WILL_FAIL TRUE)
    add_executable(rknn_runtime_tests tests/rknn_runtime_test.cpp)
    target_include_directories(rknn_runtime_tests PRIVATE "${RKNN_INCLUDE_DIR}")
    target_link_libraries(rknn_runtime_tests PRIVATE rk3576_yolo_runtime)
    add_test(NAME rknn_runtime_tests COMMAND rknn_runtime_tests)
    add_test(NAME rk3576_probe_missing_args COMMAND rk3576_yolo_probe --mode detect)
    add_test(NAME rk3576_probe_invalid_roi COMMAND rk3576_yolo_probe --model unused --roi 0 0 -1 1)
    set_tests_properties(rk3576_probe_missing_args rk3576_probe_invalid_roi PROPERTIES WILL_FAIL TRUE)
endif()

add_library(rk3576_tracker_assembly STATIC tracker_assembly.cpp platform_factory.cpp)
find_package(JPEG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VIDEO_AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
add_library(rk3576_video STATIC mjpeg_video_reader.cpp video_timing.cpp video_frame_source.cpp)
target_link_libraries(rk3576_video PUBLIC rk3576_yolo_host PRIVATE JPEG::JPEG PkgConfig::VIDEO_AV)
target_link_libraries(rk3576_tracker_assembly PUBLIC rk3576_yolo_host rk3576_video)
add_library(platform_rk3576 STATIC production_factory.cpp)
target_include_directories(platform_rk3576 PRIVATE "${RKNN_INCLUDE_DIR}")
target_link_libraries(platform_rk3576 PUBLIC rk3576_tracker_assembly rk3576_yolo_sdk)
add_subdirectory(../../replay replay)
add_executable(rk3576_tracker_replay ../../replay/rk3576_tracker_replay.cpp)
target_link_libraries(rk3576_tracker_replay PRIVATE replay_common platform_rk3576)
if(DRONE_TRACKER_BUILD_TESTS)
    add_executable(rk3576_assembly_tests tests/tracker_assembly_test.cpp)
    target_include_directories(rk3576_assembly_tests PRIVATE "${RKNN_INCLUDE_DIR}")
    target_link_libraries(rk3576_assembly_tests PRIVATE platform_rk3576)
    add_test(NAME rk3576_assembly_tests COMMAND rk3576_assembly_tests)
    add_test(NAME tracker_replay_no_args COMMAND rk3576_tracker_replay)
    add_test(NAME tracker_replay_extra_args COMMAND rk3576_tracker_replay unused.yaml --fps 60)
    set_tests_properties(tracker_replay_no_args tracker_replay_extra_args PROPERTIES WILL_FAIL TRUE)
endif()
if(DRONE_TRACKER_BUILD_TESTS)
    add_test(NAME tracker_replay_exit_codes COMMAND ${CMAKE_COMMAND}
        -DPROGRAM=$<TARGET_FILE:rk3576_tracker_replay> -P ${CMAKE_CURRENT_SOURCE_DIR}/../../replay/tests/command_test.cmake)
endif()

if(DRONE_TRACKER_BUILD_TESTS)
    set(VIDEO_FIXTURES "${CMAKE_CURRENT_BINARY_DIR}/video-fixtures")
    add_executable(video_fixture_generator tests/video_fixture_generator.cpp)
    target_link_libraries(video_fixture_generator PRIVATE rk3576_video JPEG::JPEG PkgConfig::VIDEO_AV)
    add_test(NAME video_generate_fixtures COMMAND video_fixture_generator "${VIDEO_FIXTURES}")
    set_tests_properties(video_generate_fixtures PROPERTIES FIXTURES_SETUP video_files)
    foreach(video_test mjpeg_video_reader video_timing video_frame_source)
        add_executable(${video_test}_tests tests/${video_test}_test.cpp)
        target_link_libraries(${video_test}_tests PRIVATE rk3576_video JPEG::JPEG PkgConfig::VIDEO_AV)
        add_test(NAME ${video_test}_tests COMMAND ${video_test}_tests "${VIDEO_FIXTURES}")
        set_tests_properties(${video_test}_tests PROPERTIES FIXTURES_REQUIRED video_files TIMEOUT 20)
    endforeach()
    target_compile_definitions(rk3576_assembly_tests PRIVATE VIDEO_FIXTURES="${VIDEO_FIXTURES}")
    set_tests_properties(rk3576_assembly_tests PROPERTIES FIXTURES_REQUIRED video_files)
    add_executable(video_source_driver tests/video_source_driver.cpp)
    target_link_libraries(video_source_driver PRIVATE rk3576_tracker_assembly)
    add_executable(video_board_validation tests/video_board_validation.cpp)
    target_link_libraries(video_board_validation PRIVATE rk3576_tracker_assembly)
    add_test(NAME video_factory_normal COMMAND video_source_driver "${VIDEO_FIXTURES}/420.mov" 1 0 2)
    add_test(NAME video_factory_slow COMMAND video_source_driver "${VIDEO_FIXTURES}/slow.mov" 0 60 1)
    set_tests_properties(video_factory_normal video_factory_slow PROPERTIES FIXTURES_REQUIRED video_files TIMEOUT 20)
endif()
