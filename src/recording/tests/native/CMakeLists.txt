cmake_minimum_required(VERSION 3.16)
project(recording_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
foreach(step configure build install)
    if(step STREQUAL configure)
        execute_process(COMMAND ${CMAKE_COMMAND} -S ${QUADROTOR_SOURCE} -B ${CMAKE_BINARY_DIR}/messages
            -DCMAKE_INSTALL_PREFIX=${CMAKE_BINARY_DIR}/message-install -DCMAKE_BUILD_TYPE=Release
            RESULT_VARIABLE result)
    elseif(step STREQUAL build)
        execute_process(COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}/messages -j 8 RESULT_VARIABLE result)
    else()
        execute_process(COMMAND ${CMAKE_COMMAND} --install ${CMAKE_BINARY_DIR}/messages RESULT_VARIABLE result)
    endif()
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Native real shared-message ${step} failed")
    endif()
endforeach()
list(PREPEND CMAKE_PREFIX_PATH ${CMAKE_BINARY_DIR}/message-install)
foreach(package sensor_msgs geometry_msgs nav_msgs vision_msgs std_msgs mavros_msgs quadrotor_msgs rosbag2_cpp)
    find_package(${package} REQUIRED)
endforeach()
add_library(cpu_affinity STATIC ${DRONE_REPO_ROOT}/src/platform/cpu_affinity.cpp)
target_include_directories(cpu_affinity PUBLIC ${DRONE_REPO_ROOT}/src)
target_compile_definitions(cpu_affinity PRIVATE DRONE_TRACKER_RK3576)
target_link_libraries(cpu_affinity PUBLIC yaml-cpp Threads::Threads)
add_subdirectory(${DRONE_REPO_ROOT}/src/recording recording)
add_executable(recording_transport_native_tests ../transport_test.cpp)
target_link_libraries(recording_transport_native_tests PRIVATE drone_recording)
add_executable(recording_writer_native_tests ../writer_test.cpp)
target_link_libraries(recording_writer_native_tests PRIVATE drone_recording)

add_executable(recording_affinity_native_tests ../affinity_test.cpp)
target_link_libraries(recording_affinity_native_tests PRIVATE drone_recording)
