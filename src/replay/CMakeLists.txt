find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(replay_common STATIC replay_config.cpp replay_reader.cpp replay_clock.cpp replay_report.cpp replay_runner.cpp)
target_link_libraries(replay_common PUBLIC hybrid_tracker PRIVATE OpenSSL::Crypto)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE REPLAY_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE REPLAY_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(REPLAY_DIRTY)
    set(REPLAY_DIRTY yes)
else()
    set(REPLAY_DIRTY no)
endif()
target_compile_definitions(replay_common PRIVATE REPLAY_GIT_COMMIT="${REPLAY_COMMIT}" REPLAY_GIT_DIRTY="${REPLAY_DIRTY}")
if(DRONE_TRACKER_BUILD_TESTS)
    add_executable(replay_tests tests/replay_test.cpp)
    target_link_libraries(replay_tests PRIVATE replay_common rk3576_tracker_assembly)
    target_compile_definitions(replay_tests PRIVATE REPLAY_EXAMPLE="${CMAKE_CURRENT_SOURCE_DIR}/../../parameters/replay_fp16.yaml")
    add_test(NAME replay_tests COMMAND replay_tests)
endif()
