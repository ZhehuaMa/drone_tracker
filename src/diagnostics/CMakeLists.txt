cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(drone_diagnostics LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(DRONE_TRACKER_BUILD_TESTS ON)
    enable_testing()
endif()
add_library(drone_diagnostics STATIC recorder.cpp client.cpp storage.cpp)
target_include_directories(drone_diagnostics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_definitions(drone_diagnostics PUBLIC DRONE_TRACKER_ENABLE_DIAGNOSTICS PRIVATE _GNU_SOURCE)
add_executable(rk3576_diagnostic_worker worker.cpp)
target_link_libraries(rk3576_diagnostic_worker PRIVATE drone_diagnostics)
add_executable(rk3576_diagnostic_inspect inspect.cpp)
target_link_libraries(rk3576_diagnostic_inspect PRIVATE drone_diagnostics)
install(TARGETS rk3576_diagnostic_worker rk3576_diagnostic_inspect DESTINATION lib/drone_tracker)
if(DRONE_TRACKER_BUILD_TESTS)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/target_test_program.cmake")
    drone_target_test_program(rk3576_diagnostic_worker diagnostic_worker_program)
    drone_target_test_program(rk3576_diagnostic_inspect diagnostic_inspect_program)
    add_executable(diagnostic_tests tests/diagnostic_test.cpp)
    target_link_libraries(diagnostic_tests PRIVATE drone_diagnostics pthread)
    add_test(NAME diagnostic_tests COMMAND diagnostic_tests "${diagnostic_worker_program}"
        "${diagnostic_inspect_program}")
    set_tests_properties(diagnostic_tests PROPERTIES TIMEOUT 120)
endif()
