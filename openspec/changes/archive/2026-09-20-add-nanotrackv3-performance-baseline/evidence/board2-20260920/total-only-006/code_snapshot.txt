};
inline thread_local Observer *current{nullptr};
/** @brief 只在当前线程启用观察，作用域结束后恢复原观察器。 */
class Observation final {
  public:
    explicit Observation(Observer *observer) noexcept : previous_{current} { current = observer; }
    ~Observation() noexcept { current = previous_; }
    Observation(const Observation &) = delete;
    Observation &operator=(const Observation &) = delete;

  private:
    Observer *previous_{};
};
/** @brief 轻量计时区间；显式 finish(false) 可标记普通返回值表示的失败。 */
class Span final {
  public:
    explicit Span(Stage stage) noexcept : observer_{current}, exceptions_{std::uncaught_exceptions()} {
        if (observer_) {
            id_ = observer_->begin(stage);
        }
    }
    ~Span() noexcept { finish(std::uncaught_exceptions() == exceptions_); }
    void finish(bool complete = true) noexcept {
        if (observer_) {
            observer_->end(id_, complete);
            observer_ = nullptr;
        }
    }
    Span(const Span &) = delete;
    Span &operator=(const Span &) = delete;

  private:
    Observer *observer_{};
    std::uint32_t id_{};
    int exceptions_{};
};
} // namespace timing
#endif // DRONE_TRACKER_TIMING_OBSERVER_H

int runBenchmark(const Config &config, const ModelLoader &loader) {
    std::unique_ptr<ResultWriter> writer;
    std::string status{"failed"}, reason{"preparation_error"};
    int exit_code{1};
    TimingRecorder recorder{config.detailed};
    std::size_t valid{};
    try {
        writer = std::make_unique<ResultWriter>(config);
        writer->identity["code_commit"] = BASELINE_COMMIT;
        writer->identity["build_flags"] = BASELINE_BUILD_FLAGS;
        writer->identity["compiler"] = __VERSION__;
        writer->identity["opencv"] = CV_VERSION;
        writer->identity["ffmpeg_avcodec"] = AV_STRINGIFY(LIBAVCODEC_VERSION);
        writer->identity["ffmpeg_avformat"] = AV_STRINGIFY(LIBAVFORMAT_VERSION);
        writer->identity["jpeg_version"] = std::to_string(JPEG_LIB_VERSION);
        saveText(config.output / "code_snapshot.txt", BASELINE_SOURCE_SNAPSHOT);
        writer->identity["code_snapshot_sha256"] = sha256(config.output / "code_snapshot.txt");
        writer->identity["executable_sha256"] = sha256("/proc/self/exe");
        writer->identity["video_sha256"] = sha256(config.video);
        {
            std::ifstream mappings("/proc/self/maps");
            std::string line;
            while (std::getline(mappings, line)) {
                const auto offset{line.find('/')};
                if (offset != std::string::npos && line.find("librknnrt.so", offset) != std::string::npos) {
                    const auto library{line.substr(offset)};
                    writer->identity["rknn_library_path"] = library;
                    writer->identity["rknn_library_sha256"] = sha256(library);
                    break;
                }
            }
            if (!writer->identity.count("rknn_library_path")) {
                writer->identity["rknn_library_missing_reason"] = "进程映射中没有可识别的 librknnrt.so 路径";
            }
        }
        for (std::size_t i{0}; i < config.models.size(); ++i) {
            writer->identity["model_" + std::to_string(i) + "_sha256"] = sha256(config.models[i]);
        }
        saveText(config.output / "config.yaml", readText(config.source));
        reason = "model_error";
        auto components{loader ? loader(config)
                               : rk3576::loadNanoTrack(config.models[0].string(), config.models[1].string(),
                                                       config.models[2].string())};
        if (!components) {
            throw std::runtime_error(model::describe(components.error));
        }
        writer->identity["model_attributes_runtime_driver"] = components.value.description;
        writer->diagnostic(components.value.description);
        reason = "precision_unverified";
        verifyPrecision(config, *writer, components.value.floating_interfaces);
        reason = "runtime_version_mismatch";
        if (components.value.description.find("runtime=2.3.2") == std::string::npos ||
            components.value.description.find("driver=0.9.8") == std::string::npos) {
            throw std::runtime_error("实机版本必须为 Runtime 2.3.2 / 驱动 0.9.8");
        }
        NanoTracker tracker{std::move(components.value.backend), std::move(components.value.preprocessor)};
        writer->manifest("running", "measuring");
        std::atomic<bool> stop{false};
        for (int repeat{0}; repeat <= config.repeats; ++repeat) {
            tracker.reset();
            std::unique_ptr<rk3576::MjpegVideoReader> reader;
            const bool warmup{repeat == 0};
            const int limit{warmup ? config.warmup_updates : config.max_updates};
            int updates{};
            for (int index{0};; ++index) {
                if (interrupted) {
                    status = "interrupted";
                    reason = "signal";
                    exit_code = 130;
                    break;
                }
                recorder.clear();
                writer->environment(recorder.now());
                FrameRecord frame;
                frame.repeat = repeat;
                frame.index = index;
                frame.warmup = warmup;
                frame.init = index == 0;
                reason = "read_error";
                rk3576::VideoReadResult decoded;
                bool eof{false};
                {
                    timing::Observation observe{&recorder};
                    try {
                        if (!reader) {
                            reader = std::make_unique<rk3576::MjpegVideoReader>(config.video.string(), stop);
                        }
                        decoded = reader->next();
                        frame.pts = decoded.sample.pts;
                        frame.pts_valid = decoded.sample.has_pts;
                        if (decoded.status == rk3576::VideoReadStatus::Eof) {
                            eof = true;
                        } else if (decoded.status != rk3576::VideoReadStatus::Frame) {
                            reason =
                                decoded.status == rk3576::VideoReadStatus::ReadError ? "read_error" : "decode_error";
                            throw std::runtime_error(decoded.reason);
                        } else {
                            reason = "frame_copy_error";
                            FrameRef input;
                            {
                                timing::Span copy{Stage::FrameCopy};
                                input = rk3576::copyGrayFrame(decoded.sample.gray, 0);
                            }
                            reason = "model_error";
                            if (frame.init) {
                                tracker.init(input, config.bbox);
                                recorder.finish(true);
                                if (!tracker.isInitialized()) {
                                    reason = "model_error";
                                    throw std::runtime_error("模板初始化失败");
                                }
                                frame.bbox = config.bbox;
                                frame.bbox_valid = true;
                            } else {
                                const float score{tracker.update(input, frame.bbox)};
                                recorder.finish(true);
                                if (score < 0 || !std::isfinite(score) || !finiteBox(frame.bbox)) {
                                    reason = "model_error";
                                    throw std::runtime_error("跟踪返回失败或非有限值");
                                }
                                frame.score = score;
                                frame.bbox_valid = true;
                            }
                            frame.status = "success";
                        }
                    } catch (const std::exception &error) {
                        if (recorder.active() && recorder.spans()[0].end == 0) {
                            recorder.finish(false);
                        }
                        recorder.markFailed();
                        frame.error = error.what();
                        if (reason == "read_error") {
                            reason = recorder.active() ? "decode_error" : "read_error";
                        }
                    }
                }
                if (eof) {
                    if (warmup && updates < limit) {
                        reason = "warmup_insufficient";
                        throw std::runtime_error("录像不足预热次数");
                    }
                    reason = "eof";
                    break;
                }
                writer->writeFrame(frame, recorder);
                if (frame.status != "success") {
                    throw std::runtime_error(frame.error);
                }
                if (frame.init) {
                    const auto base{reader->timeBase()};
                    writer->identity["pts_time_base"] = std::to_string(base.first) + "/" + std::to_string(base.second);
                } else {
                    ++updates;
                    if (!warmup) {
                        ++valid;
                    }
                }
                if ((warmup || limit > 0) && updates >= limit) {
                    reason = "frame_limit";
                    break;
                }
            }
            if (status == "interrupted") {
                break;
            }
        }
        if (status != "interrupted") {
            if (!valid) {
                reason = "no_valid_updates";
                throw std::runtime_error("没有有效正式更新");
            }
            status = "complete";
            exit_code = 0;
        }
    } catch (const std::ios_base::failure &error) {
        status = "failed";
        reason = "output_error";
        exit_code = 1;
        std::cerr << reason << ": " << error.what() << '\n';
    } catch (const std::exception &error) {
        status = "failed";
        exit_code = 1;
        std::cerr << reason << ": " << error.what() << '\n';
        if (writer) {
            try {
                writer->diagnostic(error.what());
            } catch (...) {
                std::cerr << "诊断输出也失败\n";
            }
        }
    }
    if (writer) {
        try {
            writer->environment(recorder.now(), true);
            const auto counted{writeReports(config.output, status, reason)};
            if (status == "complete" && counted != valid) {
                throw std::runtime_error("落盘有效帧数不一致");
            }
            writer->manifest(status, reason);
        } catch (const std::exception &error) {
            std::cerr << "output_error: " << error.what() << '\n';
            try {
                writer->manifest("failed", "output_error");
            } catch (...) {
            }
            return 1;
        }
    }
    return exit_code;
}

int integer(const cv::FileStorage &file, const char *key, int fallback, int minimum) {
    const auto node{file[key]};
    if (node.empty()) {
        return fallback;
    }
    if (!node.isInt() && !node.isReal()) {
        throw std::invalid_argument(std::string(key) + " 必须为整数");
    }
    const double value{static_cast<double>(node)};
    if (!std::isfinite(value) || value < minimum || value > std::numeric_limits<int>::max() ||
        std::floor(value) != value) {
        throw std::invalid_argument(std::string(key) + " 数值非法");
    }
    return static_cast<int>(value);
}
} // namespace
Config loadConfig(const std::filesystem::path &path) {
    Config config;
    config.source = std::filesystem::absolute(path).lexically_normal();
    cv::FileStorage file(config.source.string(), cv::FileStorage::READ);
    if (!file.isOpened()) {
        throw std::invalid_argument("配置无法打开");
    }
    if (integer(file, "schema_version", 0, 1) != 1) {
        throw std::invalid_argument("schema_version 必须为 1");
    }
    const auto resolve{[&](const char *key, bool input) {
        const auto node{file[key]};
        if (!node.isString() || node.string().empty()) {
            throw std::invalid_argument(std::string(key) + " 缺失");
        }
        std::filesystem::path result{node.string()};
        if (result.is_relative()) {
            result = config.source.parent_path() / result;
        }
        result = result.lexically_normal();
        if (input && !std::filesystem::is_regular_file(result)) {
            throw std::invalid_argument(std::string(key) + " 必须为本地普通文件: " + result.string());
        }
        return result;
    }};
    config.video = resolve("video", true);
    config.models = {resolve("template_model", true), resolve("search_model", true), resolve("head_model", true)};
    config.evidence = resolve("precision_evidence", false);
    config.output = resolve("output", false);
    if (std::filesystem::exists(config.output)) {
        throw std::invalid_argument("输出目录已存在");
    }
    const auto box{file["bbox"]};
    if (!box.isSeq() || box.size() != 4) {
        throw std::invalid_argument("bbox 必须为四个有限数值");
    }
    std::array<float, 4> values{};
    for (int i{0}; i < 4; ++i) {
        const auto node{box[i]};
        const double value{static_cast<double>(node)};
        if ((!node.isInt() && !node.isReal()) || !std::isfinite(value) ||
            std::abs(value) > std::numeric_limits<float>::max()) {
            throw std::invalid_argument("bbox 非法");
        }
        values[i] = static_cast<float>(value);
    }
    if (values[2] <= 0 || values[3] <= 0) {
        throw std::invalid_argument("bbox 宽高必须为正");
    }
    config.bbox = {values[0], values[1], values[2], values[3]};
    config.warmup_updates = integer(file, "warmup_updates", 30, 0);
    config.max_updates = integer(file, "max_updates", 0, 0);
    config.repeats = integer(file, "repeats", 3, 1);
    config.environment_interval_ms = integer(file, "environment_interval_ms", 1000, 1);
    config.detailed = integer(file, "detailed_observation", 1, 0) != 0;
    if (integer(file, "detailed_observation", 1, 0) > 1) {
        throw std::invalid_argument("detailed_observation 仅为 0/1");
    }
    config.board = file["board"].string();
    if (config.board.empty()) {
        throw std::invalid_argument("board 必须说明板卡编号");
    }
    return config;
}

void verifyPrecision(const Config &config, ResultWriter &writer, bool floating_interfaces) {
    writer.identity["precision_status"] = "precision_unverified";
    writer.identity["precision_evidence_sha256"] = sha256(config.evidence);
    std::filesystem::copy_file(config.evidence, config.output / "precision_evidence.yaml");
    if (!floating_interfaces) {
        throw std::runtime_error("precision_unverified: 模型含整数接口");
    }
    cv::FileStorage evidence(config.evidence.string(), cv::FileStorage::READ);
    if (!evidence.isOpened() || int(evidence["schema_version"]) != 1) {
        throw std::runtime_error("precision_unverified: 证据格式非法");
    }
    const auto models{evidence["models"]};
    if (!models.isSeq() || models.size() != 3) {
        throw std::runtime_error("precision_unverified: 需要三份模型证据");
    }
    const std::array<std::string, 3> roles{"template_backbone", "search_backbone", "head"};
    std::set<std::string> seen;
    for (const auto &entry : models) {
        const std::string role{entry["role"].string()};
        const auto match{std::find(roles.begin(), roles.end(), role)};
        if (match == roles.end() || !seen.insert(role).second) {
            throw std::runtime_error("precision_unverified: 角色非法");
        }
        const auto index{static_cast<std::size_t>(match - roles.begin())};
        const auto hash{sha256(config.models[index])};
        writer.identity[role + "_sha256"] = hash;
        if (entry["sha256"].string() != hash || entry["precision"].string() != "FP16") {
            throw std::runtime_error("precision_unverified: 模型指纹不符或非纯 FP16");
        }
        std::filesystem::path record{entry["export_record"].string()};
        if (record.empty()) {
            throw std::runtime_error("precision_unverified: 导出记录缺失");
        }
        if (record.is_relative()) {
            record = config.evidence.parent_path() / record;
        }
        const auto record_hash{sha256(record)};
        if (entry["export_record_sha256"].string() != record_hash) {
            throw std::runtime_error("precision_unverified: 导出记录指纹不符");
        }
        cv::FileStorage exported(record.string(), cv::FileStorage::READ);
        if (!exported.isOpened() || exported["model_sha256"].string() != hash ||
            exported["precision"].string() != "FP16" || exported["role"].string() != role ||
            exported["status"].string() != "success" || exported["source_record"].string().empty()) {
            throw std::runtime_error("precision_unverified: 导出记录与清单矛盾或缺少来源");
        }
        std::filesystem::path source{exported["source_record"].string()};
        if (source.is_relative()) {
            source = record.parent_path() / source;
        }
        if (sha256(source) != exported["source_record_sha256"].string()) {
            throw std::runtime_error("precision_unverified: 原始导出证据指纹不符");
        }
        std::filesystem::copy_file(record, config.output / (role + "_export.yaml"));
        std::filesystem::copy_file(source, config.output / (role + "_source_record.txt"));
        writer.identity[role + "_export_sha256"] = record_hash;
    }
    writer.identity["precision_status"] = "verified_export_evidence";
}

std::string jsonString(const std::string &value) {
    std::ostringstream out;
    out << '"';
    for (const unsigned char c : value) {
        if (c == '"' || c == '\\') {
            out << '\\' << c;
        } else if (c < 32) {
            out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << static_cast<int>(c);
        } else {
            out << c;
        }
    }
    out << '"';
    return out.str();
}

        }
        result += c;
    }
    return result + '"';
}

std::string readText(const std::filesystem::path &path) {
    std::ifstream input(path, std::ios::binary);
    if (!input) {
        throw std::runtime_error("读取失败: " + path.string());
    }
    std::ostringstream out;
    out << input.rdbuf();
    if (input.bad()) {
        throw std::runtime_error("读取失败: " + path.string());
    }
    return out.str();
}
std::string sha256(const std::filesystem::path &path) {
    std::ifstream input(path, std::ios::binary);
    std::unique_ptr<EVP_MD_CTX, decltype(&EVP_MD_CTX_free)> context{EVP_MD_CTX_new(), EVP_MD_CTX_free};
    if (!input || !context || EVP_DigestInit_ex(context.get(), EVP_sha256(), nullptr) != 1) {
        throw std::runtime_error("文件指纹初始化失败: " + path.string());
    }
    std::array<char, 65536> buffer{};
    while (input) {
        input.read(buffer.data(), buffer.size());
        if (EVP_DigestUpdate(context.get(), buffer.data(), static_cast<std::size_t>(input.gcount())) != 1) {
            throw std::runtime_error("文件指纹更新失败");
        }
    }
    if (!input.eof()) {
        throw std::runtime_error("文件指纹读取失败");
    }
    std::array<unsigned char, EVP_MAX_MD_SIZE> digest{};
    unsigned int length{};
    if (EVP_DigestFinal_ex(context.get(), digest.data(), &length) != 1) {
        throw std::runtime_error("指纹结束失败");
    }
    std::ostringstream out;
    for (unsigned int i{0}; i < length; ++i) {
        out << std::hex << std::setw(2) << std::setfill('0') << int(digest[i]);
    }
    return out.str();
}

void ResultWriter::writeFrame(const FrameRecord &frame, const TimingRecorder &recorder) {
    if (!recorder.valid()) {
        throw std::runtime_error("计时区间错误或缓冲容量不足");
    }
    const std::string phase{frame.warmup ? "warmup" : "measure"};
    const std::string operation{frame.init ? "init" : "track"};
    std::ostringstream prefix;
    prefix << "1," << csvString(identity.at("run_id")) << "," << frame.repeat << ',' << phase << ',' << operation << ','
           << frame.index << ',';
    frames_ << prefix.str();
    if (frame.pts_valid) {
        frames_ << frame.pts;
    }
    frames_ << ',' << frame.pts_valid << ',' << frame.status << ',' << frame.bbox_valid << ',';
    for (const auto value : {frame.bbox.x, frame.bbox.y, frame.bbox.width, frame.bbox.height}) {
        if (frame.bbox_valid && std::isfinite(value)) {
            frames_ << number(value);
        }
        frames_ << ',';
    }
    if (frame.score && std::isfinite(*frame.score)) {
        frames_ << number(*frame.score);
    }
    frames_ << ',';
    std::int64_t stages{};
    for (const auto stage :
         {Stage::Decode, Stage::FrameCopy, Stage::Preprocess, Stage::Inference, Stage::Postprocess}) {
        const auto value{recorder.duration(stage)};
        if (value) {
            frames_ << *value;
            stages += *value;
        }
        frames_ << ',';
    }
    const auto total{recorder.duration(Stage::DecodeToResult)};
    if (total) {
        if (*total < stages) {
            throw std::runtime_error("顶层计时重复或超出总区间");
        }
        frames_ << *total - stages;
    }
    frames_ << ',';
    if (total) {
        frames_ << *total;
    }
    frames_ << ',' << csvString(frame.error) << ',' << (frame.init ? "input" : "prediction") << '\n';
    for (std::size_t i{0}; i < recorder.size(); ++i) {
        const auto &span{recorder.spans()[i]};
        std::int64_t children{};
        for (std::size_t j{0}; j < recorder.size(); ++j) {
            const auto &child{recorder.spans()[j]};
            if (child.parent == span.id) {
                children += child.end - child.start;
            }
        }
        if (children > span.end - span.start) {
            throw std::runtime_error("子区间超出父区间");
        }
        spans_ << prefix.str() << serial_ + span.id << ',';
        if (span.parent) {
            spans_ << serial_ + span.parent;
        }
        spans_ << ',' << stageName(span.stage) << ',' << modelRole(recorder, span) << ',' << span.start << ','
               << span.end << ',' << span.complete << ',' << span.end - span.start - children << '\n';
    }
    serial_ += recorder.size();
    frames_.flush();
    spans_.flush();
}
void ResultWriter::environment(std::int64_t now, bool force) {
    if (!force && last_environment_ >= 0 &&
        now - last_environment_ < std::int64_t(config_.environment_interval_ms) * 1000000) {
        return;
    }
    last_environment_ = now;
    const auto sample{[&](const std::filesystem::path &path) {
        std::string value, reason;
        try {
            value = readText(path);
        } catch (const std::exception &error) {
            reason = error.what();
        }
        environment_ << "1," << now << ',' << csvString(path.string()) << ',' << csvString(value) << ','
                     << csvString(reason) << '\n';
    }};
    for (const auto &root : {"/sys/class/thermal", "/sys/devices/system/cpu/cpufreq", "/sys/class/devfreq"}) {
        std::error_code error;
        std::filesystem::directory_iterator entries(root, error);
        if (error) {
            environment_ << "1," << now << ',' << csvString(root) << ",," << csvString(error.message()) << '\n';
            continue;
        }
        for (const auto &entry : entries) {
            const std::string name{entry.path().filename().string()};
            if (name.rfind("thermal_zone", 0) == 0) {
                sample(entry.path() / "type");
                sample(entry.path() / "temp");
            } else if (name.rfind("policy", 0) == 0) {
                for (const char *field :
                     {"scaling_governor", "scaling_cur_freq", "scaling_min_freq", "scaling_max_freq"}) {
                    sample(entry.path() / field);
                }
            } else if (std::string(root) == "/sys/class/devfreq") {
                for (const char *field : {"governor", "cur_freq", "min_freq", "max_freq"}) {
                    sample(entry.path() / field);
                }
            }
        }
    }
    environment_.flush();
}

bool readCsvRecord(std::istream &input, std::vector<std::string> &row) {
    row.clear();
    std::string field;
    bool quoted{false}, started{false};
    char c{};
    while (input.get(c)) {
        started = true;
        if (c == '"') {
            if (quoted && input.peek() == '"') {
                input.get(c);
                field += '"';
            } else {
                quoted = !quoted;
            }
        } else if (c == ',' && !quoted) {
            row.push_back(field);
            field.clear();
        } else if (c == '\n' && !quoted) {
            row.push_back(field);
            return true;
        } else {
            field += c;
        }
    }
    if (quoted || started || input.bad()) {
        throw std::runtime_error("CSV 尾行不完整或读取失败");
    }
    return false;
}
Distribution distribution(std::vector<std::int64_t> samples) {
    Distribution result;
    result.count = samples.size();
    if (samples.empty()) {
        return result;
    }
    std::sort(samples.begin(), samples.end());
    long double total{};
    for (const auto value : samples) {
        total += value;
    }
    const long double mean{total / samples.size()};
    long double squared{};
    for (const auto value : samples) {
        squared += (value - mean) * (value - mean);
    }
    result.mean = static_cast<double>(mean);
    result.standard_deviation = static_cast<double>(std::sqrt(squared / samples.size()));
    result.minimum = samples.front();
    result.maximum = samples.back();
    const auto percentile{
        [&](double p) { return samples.at(static_cast<std::size_t>(std::ceil(p * samples.size())) - 1); }};
    result.p50 = percentile(0.50);
    result.p90 = percentile(0.90);
    result.p95 = percentile(0.95);
    result.p99 = percentile(0.99);
    return result;
}

void readRows(const std::filesystem::path &path, const std::function<void(const CsvRow &)> &consume) {
    std::ifstream file(path);
    if (!file) {
        throw std::runtime_error("统计输入无法打开");
    }
    std::vector<std::string> headers, values;
    if (!readCsvRecord(file, headers)) {
        throw std::runtime_error("CSV 表头缺失");
    }
    while (readCsvRecord(file, values)) {
        if (values.size() != headers.size()) {
            throw std::runtime_error("CSV 列数不匹配");
        }
        CsvRow row;
        for (std::size_t i{0}; i < headers.size(); ++i) {
            row[headers[i]] = values[i];
        }
        consume(row);
    }
}

                    "|---|---:|---:|---:|---:|---:|---:|---:|---:|---:|\n";
        bool first_metric{true};
        for (const auto &metric : group.second) {
            const auto d{distribution(metric.second)};

                json << ',';
            }

            json << jsonString(metric.first) << ":{\"count\":" << d.count;
            markdown << "| " << metric.first << " | " << d.count;

            }

            markdown << " |\n";
        }

        const auto &total{group.second.at("decode_to_result_ns")};
        const bool tracks{group.first.substr(group.first.find('/') + 1) == "track"};
        const long double sum{std::accumulate(total.begin(), total.end(), 0.0L)};
        if (tracks && !total.empty() && sum > 0) {
            const double fps{static_cast<double>(total.size() * 1e9L / sum)};
            json << number(fps);
            markdown << "\n解码至结果处理速率：" << number(fps) << " 帧/秒。\n";
            if (group.first.rfind("all/", 0) != 0) {
                repeat_means.push_back(static_cast<double>(sum / total.size()));
            }
        } else {
            json << "null";
        }
        json << '}';
    }
    json << "},\"repeat_update_mean_range_ns\":";
    if (repeat_means.empty()) {
        json << "null";
    } else {
        const auto range{std::minmax_element(repeat_means.begin(), repeat_means.end())};
        json << '[' << jsonString(number(*range.first)) << ',' << jsonString(number(*range.second)) << ']';
        markdown << "\n各轮更新均值范围：" << number(*range.first) << "～" << number(*range.second) << " 纳秒。\n";
    }
    json << "}\n";
    json.close();
    markdown.close();
    return valid;
}

namespace {
using namespace baseline;
using Samples = std::map<std::string, std::map<std::string, std::vector<std::int64_t>>>;
void check(bool ok, const std::string &message) {
    if (!ok) {
        throw std::runtime_error(message);
    }
}
std::vector<CsvRow> rows(const std::filesystem::path &path) {
    std::ifstream input(path);
    check(bool(input), "无法打开 " + path.string());
    std::vector<std::string> header, cells;
    check(readCsvRecord(input, header), "表头缺失");
    std::vector<CsvRow> result;
    while (readCsvRecord(input, cells)) {
        check(cells.size() == header.size(), "列数错误");
        CsvRow row;
        for (std::size_t i{0}; i < cells.size(); ++i) {
            row.emplace(header[i], cells[i]);
        }
        result.push_back(std::move(row));
    }
    return result;
}
std::string key(const CsvRow &row) { return row.at("repeat_id") + "/" + row.at("phase") + "/" + row.at("frame_index"); }
} // namespace
int main(int argc, char **argv) {
    try {
        check(argc == 2 || argc == 3, "用法：baseline_verify_records RUN_DIRECTORY [REFERENCE_DIRECTORY]");
        const std::filesystem::path root{argv[1]};
        const auto frames{rows(root / "frames.csv")};
        const auto spans{rows(root / "spans.csv")};
        // OpenCV 4.5 的 JSON 读取器不接受标准 null；仅在内存解析副本中替换该字面量。
        const auto raw_json{readText(root / "summary.json")};
        std::string parse_json;
        bool quoted{false}, escaped{false};
        for (std::size_t i{0}; i < raw_json.size(); ++i) {
            const char c{raw_json[i]};
            if (!quoted && raw_json.compare(i, 4, "null") == 0) {
                parse_json += "\"__null__\"";
                i += 3;
                continue;
            }
            parse_json += c;
            if (c == '\\' && quoted && !escaped) {
                escaped = true;
                continue;
            }
            if (c == '\"' && !escaped) {
                quoted = !quoted;
            }
            escaped = false;
        }
        cv::FileStorage summary(parse_json,
                                cv::FileStorage::READ | cv::FileStorage::MEMORY | cv::FileStorage::FORMAT_JSON);
        check(summary.isOpened(), "JSON 汇总无法打开");
        Samples samples;
        std::map<std::string, const CsvRow *> frame_index;
        std::map<std::string, const CsvRow *> span_index;
        std::map<std::string, std::vector<const CsvRow *>> child_index;
        std::map<std::string, std::map<std::string, int>> calls;
        std::size_t valid{}, failed{}, warmup{}, init{};
        for (const auto &frame : frames) {
            check(frame_index.emplace(key(frame), &frame).second, "重复帧身份");
            if (frame.at("phase") != "measure") {
                ++warmup;
                continue;
            }
            const std::string kind{frame.at("status") == "success" ? frame.at("operation") : "failed"};
            if (kind == "track") {
                ++valid;
            } else if (kind == "init") {
                ++init;
            } else {
                ++failed;
            }
            for (const auto &field : frame) {
                if (field.first.size() < 3 || field.first.substr(field.first.size() - 3) != "_ns") {
                    continue;
                }
                for (const auto &group : {frame.at("repeat_id") + "/" + kind, "all/" + kind}) {
                    auto &values{samples[group][field.first]};
                    if (!field.second.empty()) {
                        values.push_back(std::stoll(field.second));
                    }
                }
            }
            if (!frame.at("decode_to_result_ns").empty()) {
                std::int64_t sum{};
                for (const auto &field :
                     {"decode_ns", "frame_copy_ns", "preprocess_ns", "inference_ns", "postprocess_ns", "other_ns"}) {
                    if (!frame.at(field).empty()) {
                        sum += std::stoll(frame.at(field));
                    }
                }
                check(sum == std::stoll(frame.at("decode_to_result_ns")), "顶层阶段与总计时不相等");
            }
            if (kind == "init") {
                check(frame.at("score").empty() && frame.at("postprocess_ns").empty(), "初始化伪造分数或后处理");
            }
        }
        for (const auto &span : spans) {
            check(span_index.emplace(span.at("span_id"), &span).second, "重复区间 ID");
            child_index[span.at("parent_id")].push_back(&span);
        }
        for (const auto &span : spans) {
            const auto start{std::stoll(span.at("start_ns"))}, end{std::stoll(span.at("end_ns"))};
            check(end >= start, "负计时");
            const auto parent{span.at("parent_id")};
            if (!parent.empty()) {
                check(span_index.count(parent) == 1, "父区间缺失");
                const auto &p{*span_index.at(parent)};
                check(key(p) == key(span) && std::stoll(p.at("start_ns")) <= start && std::stoll(p.at("end_ns")) >= end,
                      "区间不属于父帧或超出父区间");
            }
            std::int64_t children{};
            std::int64_t previous_end{start};
            for (const auto *child_pointer : child_index[span.at("span_id")]) {
                const auto &child{*child_pointer};
                {
                    check(std::stoll(child.at("start_ns")) >= previous_end, "同层区间重叠");
                    previous_end = std::stoll(child.at("end_ns"));
                    children += previous_end - std::stoll(child.at("start_ns"));
                }
            }
            check(end - start - children == std::stoll(span.at("residual_ns")), "residual 不一致");
            const auto &frame{*frame_index.at(key(span))};
            if (span.at("stage") == "template_backbone" || span.at("stage") == "search_backbone" ||
                span.at("stage") == "head") {
                ++calls[key(span)][span.at("stage")];
                check(span.at("model_role") == span.at("stage"), "模型角色错误");
            }
            if (frame.at("phase") != "measure") {
                continue;
            }
            const std::string kind{frame.at("status") == "success" ? frame.at("operation") : "failed"};
            const auto metric{"span/" + span.at("model_role") + "/" + span.at("stage")};
            for (const auto &group : {frame.at("repeat_id") + "/" + kind, "all/" + kind}) {
                samples[group][metric].push_back(end - start);
                samples[group][metric + "/residual"].push_back(end - start - children);
            }
        }
        if (!calls.empty()) {
            for (const auto &frame : frames) {
                if (frame.at("status") != "success") {
                    continue;
                }
                const auto &c{calls[key(frame)]};
                const auto count{[&](const char *name) {
                    const auto it{c.find(name)};
                    return it == c.end() ? 0 : it->second;
                }};
                const bool initialization{frame.at("operation") == "init"};
                check(count("template_backbone") == (initialization ? 1 : 0) &&
                          count("search_backbone") == (initialization ? 0 : 1) &&
                          count("head") == (initialization ? 0 : 1),
                      "模型调用次数不符");
            }
        }
        check(int(summary["valid_updates"]) == static_cast<int>(valid) &&
                  int(summary["failed_frames"]) == static_cast<int>(failed) &&
                  int(summary["initializations"]) == static_cast<int>(init) &&
                  int(summary["warmup_frames"]) == static_cast<int>(warmup),
              "汇总帧数不一致");
        std::size_t metrics{};
        for (const auto &group : summary["groups"]) {
            const auto name{group.name()};
            for (const auto &metric : group["metrics"]) {
                auto values{samples[name][metric.name()]};
                std::sort(values.begin(), values.end());
                check(int(metric["count"]) == static_cast<int>(values.size()), "统计样本数不同");
                if (values.empty()) {
                    continue;
                }
                const long double sum{std::accumulate(values.begin(), values.end(), 0.0L)};
                const long double mean{sum / values.size()};
                long double variance{};
                for (const auto value : values) {
                    variance += (value - mean) * (value - mean);
                }
                const auto compare{[&](const char *field, long double expected) {
                    const long double actual{std::stold(metric[field].string())};
                    check(std::abs(actual - expected) <= std::max(0.001L, std::abs(expected) * 1e-12L),
                          "统计重算不符: " + name + "/" + metric.name() + "/" + field);
                }};
                compare("mean", mean);
                compare("standard_deviation", std::sqrt(variance / values.size()));
                compare("minimum", values.front());
                compare("maximum", values.back());
                for (const auto &p : std::array<std::pair<const char *, int>, 4>{
                         {{"p50", 50}, {"p90", 90}, {"p95", 95}, {"p99", 99}}}) {
                    compare(p.first, values[(values.size() * p.second + 99) / 100 - 1]);
                }
                ++metrics;
            }
            if (name.substr(name.find('/') + 1) == "track") {
                const auto &v{samples[name]["decode_to_result_ns"]};
                const long double total{std::accumulate(v.begin(), v.end(), 0.0L)};
                if (total > 0) {
                    check(std::abs(double(group["decode_to_result_processing_fps"]) - v.size() * 1e9L / total) < 1e-9,
                          "速率重算不符");
                }
            }
        }
        double max_box{}, max_score{};
        if (argc == 3) {
            const auto reference{rows(std::filesystem::path(argv[2]) / "frames.csv")};
            check(reference.size() == frames.size(), "对照帧数不同");
            for (std::size_t i{0}; i < frames.size(); ++i) {
                check(key(frames[i]) == key(reference[i]) && frames[i].at("status") == reference[i].at("status"),
                      "对照身份不同");
                for (const auto *field : {"x", "y", "width", "height", "score"}) {
                    if (frames[i].at(field).empty()) {
                        check(reference[i].at(field).empty(), "有效性不同");
                        continue;
                    }
                    const double delta{std::abs(std::stod(frames[i].at(field)) - std::stod(reference[i].at(field)))};
                    if (std::string(field) == "score") {
                        max_score = std::max(max_score, delta);
                    } else {
                        max_box = std::max(max_box, delta);
                    }
                }
            }
            check(max_box <= 1e-4 && max_score <= 1e-6, "真实有无观测数值超过容差");
        }
        std::cout << "{\"status\":\"passed\",\"frames\":" << frames.size() << ",\"spans\":" << spans.size()
                  << ",\"metrics_checked\":" << metrics << ",\"max_bbox_error\":" << number(max_box)
                  << ",\"max_score_error\":" << number(max_score) << "}\n";
    } catch (const std::exception &error) {
        std::cerr << error.what() << '\n';
        return 1;
    }
}

namespace baseline {

const std::int64_t process_origin{TimingRecorder::steadyNow()};

std::int64_t TimingRecorder::steadyNow() noexcept {
    return std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch())
        .count();
}
TimingRecorder::TimingRecorder(bool detailed, Clock clock) noexcept
    : detailed_{detailed}, clock_{clock}, origin_{clock == steadyNow ? process_origin : clock()} {}
std::uint32_t TimingRecorder::begin(Stage stage) noexcept {
    if (stage == Stage::Decode && size_ == 0) {
        spans_[0] = {1, 0, Stage::DecodeToResult, now(), 0, false};
        size_ = 1;
        top_ = 1;
    }
    if (!detailed_) {
        return 0;
    }
    if (!top_ || size_ == spans_.size()) {
        error_ = true;
        return 0;
    }
    const auto id{static_cast<std::uint32_t>(size_ + 1)};
    spans_[size_++] = {id, top_, stage, now(), 0, false};
    top_ = id;
    return id;
}
void TimingRecorder::end(std::uint32_t id, bool complete) noexcept {
    if (!id) {
        return;
    }
    if (id != top_ || id > size_) {
        error_ = true;
        return;
    }
    auto &span{spans_[id - 1]};
    span.end = now();
    span.complete = complete;
    if (span.end < span.start) {
        error_ = true;
    }
    top_ = span.parent;
}
void TimingRecorder::finish(bool complete) noexcept {
    if (!active()) {
        return;
    }
    if (top_ != 1) {
        error_ = true;
    }
    while (top_ > 1) {
        end(top_, false);
    }
    end(1, complete && !error_);
}
void TimingRecorder::clear() noexcept {
    size_ = 0;
    top_ = 0;
    error_ = false;
}
std::optional<std::int64_t> TimingRecorder::duration(Stage stage) const {
    std::optional<std::int64_t> value;
    for (std::size_t i{0}; i < size_; ++i) {
        const auto &span{spans_[i]};
        if (span.stage == stage) {
            value = value.value_or(0) + span.end - span.start;
        }
    }
    return value;
}

                                                 "input_prepare",

                                                 "run",

const char *modelRole(const TimingRecorder &recorder, const SpanRecord &span) {
    auto id{span.id};
    while (id) {
        const auto &parent{recorder.spans()[id - 1]};
        if (parent.stage == Stage::TemplateBackbone || parent.stage == Stage::SearchBackbone ||
            parent.stage == Stage::Head) {
            return stageName(parent.stage);
        }
        id = parent.parent;
    }
    return "";
}

#ifndef NANOTRACK_BASELINE_TIMING_RECORDER_H
#define NANOTRACK_BASELINE_TIMING_RECORDER_H
#include "timing_observer.h"
#include <array>
#include <cstdint>
#include <optional>
namespace baseline {
using Stage = timing::Stage;
struct SpanRecord {
    std::uint32_t id{}, parent{};
    Stage stage{};
    std::int64_t start{}, end{};
    bool complete{};
};
/** @brief 每帧固定容量记录器；满容量或区间错误导致本帧失败，不丢弃后继续测量。 */
class TimingRecorder final : public timing::Observer {
  public:
    using Clock = std::int64_t (*)() noexcept;
    explicit TimingRecorder(bool detailed = true, Clock clock = steadyNow) noexcept;
    static std::int64_t steadyNow() noexcept;
    std::uint32_t begin(Stage stage) noexcept override;
    void end(std::uint32_t id, bool complete) noexcept override;
    void finish(bool complete) noexcept;
    void clear() noexcept;
    void markFailed() noexcept {
        if (size_) {
            spans_[0].complete = false;
        }
    }
    std::optional<std::int64_t> duration(Stage stage) const;
    const auto &spans() const noexcept { return spans_; }
    std::size_t size() const noexcept { return size_; }
    bool valid() const noexcept { return !error_; }
    bool active() const noexcept { return size_ != 0; }
    std::int64_t now() const noexcept { return clock_() - origin_; }

  private:
    std::array<SpanRecord, 256> spans_{};
    std::size_t size_{};
    std::uint32_t top_{};
    bool detailed_{}, error_{};
    Clock clock_{};
    std::int64_t origin_{};
};
const char *stageName(Stage stage);
