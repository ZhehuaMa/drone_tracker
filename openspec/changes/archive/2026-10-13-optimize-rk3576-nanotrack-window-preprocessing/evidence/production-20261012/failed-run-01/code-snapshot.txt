    for (const auto &directory : {"/sys/devices/system/cpu/cpufreq", "/sys/class/devfreq", "/sys/class/thermal"}) {
        if (!std::filesystem::is_directory(directory)) { continue; }
        for (const auto &entry : std::filesystem::directory_iterator{directory}) {
            for (const auto &name : {"scaling_cur_freq", "scaling_governor", "cur_freq", "governor", "temp", "type"}) {
                const auto path{entry.path() / name};

                    try { result << path.string() << '=' << read(path); }
                    catch (const std::exception &error) { result << error.what() << '\n'; }
                }
            }
        }
    }
    return result.str();
}

std::unique_ptr<NanoDcfTracker> tracker(const wp::Config &config, bool legacy,
                                     std::unique_ptr<::NanoTrackPreprocessor> checked = {}) {
    auto loaded{rk3576::loadNanoTrack(config.models[0].string(), config.models[1].string(), config.models[2].string())};
    if (!loaded) { throw std::runtime_error(model::describe(loaded.error)); }
    if (legacy) { loaded.value.preprocessor = std::make_unique<wp::WholeFramePreprocessor>(); }
    if (checked) { loaded.value.preprocessor = std::move(checked); }
    auto nano{std::make_unique<NanoTracker>(std::move(loaded.value.backend), std::move(loaded.value.preprocessor))};
    return std::make_unique<NanoDcfTracker>(config.parameters, std::move(nano));
}
std::string raw(const std::vector<wp::Batch> &batches) {

std::string intervals(const std::vector<wp::Batch> &batches) {
    std::ostringstream file;
    file << "warmup,pair,order,legacy,frame,id,parent,stage,start_ns,end_ns,complete\n";
    for (const auto &batch : batches) {
        for (std::size_t i{0}; i < batch.frames.size(); ++i) {
            const auto &recorder{batch.frames[i].recorder};
            for (std::size_t j{0}; j < recorder.size(); ++j) {
                const auto &span{recorder.intervals()[j]};
                file << batch.warmup << ',' << batch.pair << ',' << batch.order << ',' << batch.legacy << ','
                     << i << ','
                     << span.id << ',' << span.parent << ',' << baseline::stageName(span.stage) << ','
                     << span.start << ',' << span.end << ',' << span.complete << '\n';
            }
        }
    }
    return file.str();
}
std::string windowSamples(const FrameRef &frame, int warmups, int pairs) {
    std::ostringstream output;
    output << "side,warmup,pair,order,legacy,template_ns,search_ns,exact\n";
    const ModelInputContract template_contract{{1, 3, 127, 127}}, search_contract{{1, 3, 255, 255}};
    // 此诊断只测预处理，固定同一真实图像与中心；不运行模型或修改恢复初始框。
    for (const int side : {1, 32, 77, 154, 255, 511, 1024, 2048}) {
        wp::WholeFramePreprocessor old;
        rk3576::NanoTrackPreprocessor current;
        std::array<std::vector<float>, 2> expected;
        for (const auto &tag : wp::plan(warmups, pairs)) {
            auto &preprocessor{tag.legacy ? static_cast<::NanoTrackPreprocessor &>(old)
                                         : static_cast<::NanoTrackPreprocessor &>(current)};
            preprocessor.reset();
            std::array<std::int64_t, 2> elapsed{};
            for (int role{0}; role < 2; ++role) {
                const auto start{wp::Recorder::now()};
                auto prepared{role == 0 ? preprocessor.prepareTemplate(frame, {776.045f, 478.8966f},
                    side, template_contract) : preprocessor.prepareSearch(frame, {776.045f, 478.8966f},
                    side, search_contract)};
                elapsed[role] = wp::Recorder::now() - start;
                if (!prepared) { throw std::runtime_error("其他窗口预处理失败"); }
                const auto input{std::dynamic_pointer_cast<const rk3576::FloatInput>(prepared.value)};
                if (!input) { throw std::runtime_error("其他窗口缺少 Float32 结果"); }
                if (tag.order == 0) { expected[role] = *input->values(); }
                else if (expected[role] != *input->values()) {
                    throw std::runtime_error("其他窗口逐元素对照失败");
                }
            }
            output << side << ',' << tag.warmup << ',' << tag.pair << ',' << tag.order << ','
                   << tag.legacy << ',' << elapsed[0] << ',' << elapsed[1] << ",1\n";
        }
    }
    return output.str();
}

PreparedInputResult ExactPreprocessor::checked(PreparedInputResult current,

std::int64_t Recorder::now() noexcept {
    return std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now().time_since_epoch())
        .count();
}
std::uint32_t Recorder::begin(timing::Stage stage) noexcept {
    if (size_ == intervals_.size()) { valid_ = false; return 0; }
    const auto id{static_cast<std::uint32_t>(size_ + 1)};
    intervals_[size_++] = {id, top_, stage, clock_(), 0, false};
    top_ = id;
    return id;
}
void Recorder::end(std::uint32_t id, bool complete) noexcept {
    if (!id || id != top_ || id > size_) { valid_ = false; return; }
    auto &interval{intervals_[id - 1]};
    interval.end = clock_();
    interval.complete = complete;
    valid_ = valid_ && interval.end >= interval.start;
    top_ = interval.parent;
}
std::int64_t Recorder::duration(timing::Stage stage) const {
    std::int64_t total{};
    for (std::size_t i{0}; i < size_; ++i) {
        // 同名嵌套阶段只统计最外层，其他父子阶段分别报告，禁止加成总耗时。
        const auto &span{intervals_[i]};
        if (span.stage != stage) { continue; }
        bool nested{false};
        for (auto id{span.parent}; id; id = intervals_[id - 1].parent) {
            nested = nested || intervals_[id - 1].stage == stage;
        }
        if (!nested) { total += span.end - span.start; }
    }
    return total;
}
int Recorder::count(timing::Stage stage) const {
    int total{};
    for (std::size_t i{0}; i < size_; ++i) { total += intervals_[i].stage == stage; }
    return total;
}
std::vector<BatchTag> plan(int warmups, int pairs) {
    require(warmups >= 2 && pairs >= 10 && warmups <= 10000 && pairs <= 10000, "批次数量不符合测量要求");
    std::vector<BatchTag> result;
    for (int pair{0}; pair < warmups + pairs; ++pair) {
        for (int order{0}; order < 2; ++order) {
            result.push_back({pair < warmups, (pair + order) % 2 == 0, pair, order});
        }
    }
    return result;
}
std::vector<std::int64_t> formalSamples(const std::vector<Batch> &batches, bool legacy, bool preprocessing) {
    std::vector<std::int64_t> result;
    for (const auto &batch : batches) {
        if (!batch.warmup && batch.legacy == legacy) {
            result.push_back(preprocessing ? batch.preprocess : batch.elapsed);
        }
    }
    return result;
}
Batch recover(NanoDcfTracker &tracker, const std::vector<FrameRef> &frames, const cv::Rect2f &box, bool observed) {
    require(frames.size() == 25, "恢复需要恰好 25 帧");
    std::uint64_t last{};
    for (const auto &frame : frames) {
        require(frame && frame->timestampUs() > last, "恢复帧必须按时间戳严格递增");
        last = frame->timestampUs();
    }
    tracker.reset();
    Batch batch{};
    batch.frames.resize(frames.size()); // 分配位于计时外。
    const auto start{Recorder::now()};
    for (std::size_t i{0}; i < frames.size(); ++i) {
        auto &result{batch.frames[i]};
        result.timestamp = frames[i]->timestampUs();
        result.box = i == 0 ? box : cv::Rect2f{};
        const auto begin{Recorder::now()};
        {
            timing::Observation observation{observed ? &result.recorder : nullptr};
            result.success = i == 0 ? tracker.init(frames[i], box) : tracker.update(frames[i], result.box);
        }
        result.elapsed = Recorder::now() - begin;
        result.initialized = tracker.isInitialized();
        result.score = tracker.lastScore();
        batch.preprocess += result.recorder.duration(timing::Stage::Preprocess);
    }
    batch.elapsed = Recorder::now() - start;
    for (const auto &frame : batch.frames) { require(frame.recorder.valid(), "阶段计时区间不完整"); }
    return batch;
}
void compare(const Batch &old_batch, const Batch &new_batch, bool check_calls) {
    require(old_batch.frames.size() == 25 && new_batch.frames.size() == 25, "恢复输出缺帧");
    for (std::size_t i{0}; i < 25; ++i) {
        const auto &a{old_batch.frames[i]}, &b{new_batch.frames[i]};
        require(a.timestamp == b.timestamp && a.success == b.success && a.initialized == b.initialized
            && a.box == b.box && a.score == b.score, "新旧恢复状态、框、分数或帧顺序不一致");
        require(a.success && a.initialized, "指定恢复用例未全部成功");
        if (check_calls) {
            for (const auto *frame : {&a, &b}) {
                require(frame->recorder.count(timing::Stage::TemplateBackbone) == (i == 0 ? 1 : 0)
                    && frame->recorder.count(timing::Stage::SearchBackbone) == (i == 0 ? 0 : 1)
                    && frame->recorder.count(timing::Stage::Head) == (i == 0 ? 0 : 1), "模型调用次数不符合 1/24/24");
            }
        }
    }
}
Distribution distribution(std::vector<std::int64_t> samples) {
    require(!samples.empty(), "统计样本为空");
    require(std::all_of(samples.begin(), samples.end(), [](auto value) { return value >= 0; }), "耗时不能为负");
    std::sort(samples.begin(), samples.end());
    const auto n{samples.size()};
    const double median{n % 2 ? static_cast<double>(samples[n / 2]) :
        static_cast<double>(samples[n / 2 - 1]) / 2 + static_cast<double>(samples[n / 2]) / 2};
    return {median, static_cast<double>(samples[static_cast<std::size_t>(std::ceil(0.95 * n)) - 1])};
}
std::string pixelHash(const cv::Mat &gray) {
    require(!gray.empty() && gray.type() == CV_8UC1, "像素校验仅接受 Gray8");
    std::unique_ptr<EVP_MD_CTX, decltype(&EVP_MD_CTX_free)> context{EVP_MD_CTX_new(), EVP_MD_CTX_free};
    require(context && EVP_DigestInit_ex(context.get(), EVP_sha256(), nullptr) == 1, "无法初始化像素 SHA-256");
    for (int y{0}; y < gray.rows; ++y) {
        require(EVP_DigestUpdate(context.get(), gray.ptr(y), gray.cols) == 1, "像素 SHA-256 更新失败");
    }
    std::array<unsigned char, 32> digest{};
    unsigned int size{};
    require(EVP_DigestFinal_ex(context.get(), digest.data(), &size) == 1 && size == digest.size(), "像素校验失败");
    std::ostringstream output;
    for (auto byte : digest) { output << std::hex << std::setw(2) << std::setfill('0') << static_cast<int>(byte); }
    return output.str();
}

PreparedInputResult WholeFramePreprocessor::prepare(const FrameRef &frame, const cv::Point2f &position,
                                                   int original_size, const ModelInputContract &contract,
                                                   bool is_template) {
    timing::Span image_prepare{timing::Stage::ImagePrepare};
    const auto error{validatePreprocessFrame(frame)};
    if (error != PreprocessError::None) {
        image_prepare.finish(false);
        return PreparedInputResult::failure(error, preprocessErrorName(error));
    }
    const int size{is_template ? 127 : 255};
    if (!matchesImageContract(contract, 1.0f, size)) {
        image_prepare.finish(false);
        return PreparedInputResult::failure(PreprocessError::ContractMismatch, "Expected Float32 NCHW NanoTrack input");
    }
    if (!is_template && !mean_.valid()) {
        image_prepare.finish(false);
        return PreparedInputResult::failure(PreprocessError::ContractMismatch, "Current target template is required");
    }
    const auto geometry{calculateNanoCropGeometry({frame->width(), frame->height()}, position, original_size)};
    if (!geometry) {
        image_prepare.finish(false);
        return PreparedInputResult::failure(geometry.error, geometry.message);
    }
    try {
        auto storage{cache_[is_template ? 0 : 1].acquire(
            [size] { return std::make_shared<std::vector<float>>(3 * size * size); })};
        if (!storage) {
            image_prepare.finish(false);
            return PreparedInputResult::failure(PreprocessError::InputBusy, "Previous input still owns cache lease");
        }
        if (is_template) {
            mean_.initialize(frame);
        }
        cv::Mat snapshot;
        {
            const auto view{frame->mapRead()};
            if (!view) {
                image_prepare.finish(false);
                return PreparedInputResult::failure(PreprocessError::PlatformFailure, "Frame mapping failed");
            }
            const auto &mat{view->mat()};
            if (view->width() != frame->width() || view->height() != frame->height() ||
                view->pixelFormat() != FramePixelFormat::Gray8 || mat.type() != CV_8UC1 || mat.step[1] != 1 ||
                mat.cols != frame->width() || mat.rows != frame->height() || !mat.data ||
                view->rowStrideBytes() != mat.step[0] || mat.step[0] < static_cast<std::size_t>(mat.cols) ||
                mat.step[0] > std::numeric_limits<std::size_t>::max() / static_cast<std::size_t>(mat.rows)) {
                image_prepare.finish(false);
                return PreparedInputResult::failure(PreprocessError::PlatformTypeMismatch, "Invalid Gray8 mapping");
            }
            mat.convertTo(snapshot, CV_32F);
        }
        const auto &g{geometry.value};
        cv::Mat padded;
        cv::copyMakeBorder(snapshot, padded, g.top, g.bottom, g.left, g.right, cv::BORDER_CONSTANT,
                           cv::Scalar{static_cast<double>(mean_.value())});
        const cv::Rect crop{g.crop.x + g.left, g.crop.y + g.top, original_size, original_size};
        cv::Mat plane(size, size, CV_32FC1, storage->data());
        cv::resize(padded(crop), plane, {size, size}, 0, 0, cv::INTER_LINEAR);
        const auto count{static_cast<std::size_t>(size * size)};
        std::copy_n(storage->data(), count, storage->data() + count);
        std::copy_n(storage->data(), count, storage->data() + 2 * count);
        PreparedInputRef prepared{std::make_shared<rk3576::FloatInput>(std::move(storage), contract)};
        if (is_template) {
            mean_.commit();
        }
        return {std::move(prepared)};
    } catch (const std::exception &exception) {
        image_prepare.finish(false);
        return PreparedInputResult::failure(PreprocessError::PlatformFailure, exception.what());
    }
}
} // namespace window_preprocess
