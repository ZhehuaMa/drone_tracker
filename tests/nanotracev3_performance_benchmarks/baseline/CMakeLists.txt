cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(nanotrackv3_baseline LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    option(DRONE_TRACKER_BUILD_TESTS "Build controlled baseline tests" ON)
    enable_testing()
    set(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
    include(${REPOSITORY_ROOT}/src/platform/rk3576/dependencies.cmake)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn video imgcodecs videoio)
    add_subdirectory(${REPOSITORY_ROOT}/src/hybrid_tracker common)
    add_subdirectory(${REPOSITORY_ROOT}/src/platform/rk3576 rk3576)
endif()
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(JPEG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(BASELINE_AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
include(${CMAKE_CURRENT_SOURCE_DIR}/code_identity.cmake)
add_library(nanotrack_baseline_support STATIC config.cpp timing_recorder.cpp result_writer.cpp statistics.cpp model_preparation.cpp)
target_link_libraries(nanotrack_baseline_support PUBLIC hybrid_tracker OpenSSL::Crypto)
target_include_directories(nanotrack_baseline_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(nanotrack_baseline_runner STATIC benchmark_runner.cpp)
baseline_code_identity(nanotrack_baseline_runner "${REPOSITORY_ROOT}")
target_include_directories(nanotrack_baseline_runner PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(nanotrack_baseline_runner PUBLIC nanotrack_baseline_support rk3576_yolo_sdk rk3576_video PRIVATE PkgConfig::BASELINE_AV JPEG::JPEG)
add_executable(nanotrackv3_baseline main.cpp)
target_link_libraries(nanotrackv3_baseline PRIVATE nanotrack_baseline_runner)
install(TARGETS nanotrackv3_baseline DESTINATION lib/drone_tracker)
if(DRONE_TRACKER_BUILD_TESTS)
    add_executable(nanotrack_baseline_tests tests/baseline_test.cpp)
    target_link_libraries(nanotrack_baseline_tests PRIVATE nanotrack_baseline_support rk3576_video rk3576_yolo_runtime)

    add_test(NAME nanotrack_baseline_help COMMAND nanotrackv3_baseline --help)
endif()
if(DRONE_TRACKER_BUILD_TESTS)
    foreach(name rknn_session rknn_nanotrack_backend nanotrack_preprocess)

    add_executable(baseline_video_fixtures ${REPOSITORY_ROOT}/src/platform/rk3576/tests/video_fixture_generator.cpp)
