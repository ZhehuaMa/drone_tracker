cmake_minimum_required(VERSION 3.16)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(nanotrack_window_preprocess LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    set(DRONE_TRACKER_PLATFORM rk3576)
    option(DRONE_TRACKER_BUILD_TESTS "Build controlled window tests" ON)
    enable_testing()
    include(${REPOSITORY_ROOT}/src/platform/rk3576/dependencies.cmake)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn video imgcodecs videoio)
    add_subdirectory(${REPOSITORY_ROOT}/src/hybrid_tracker common)
    add_subdirectory(${REPOSITORY_ROOT}/src/platform/rk3576 rk3576)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../baseline baseline)
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(nanotrack_window_support STATIC whole_frame_preprocessor.cpp recovery.cpp performance_verdict.cpp)
target_link_libraries(nanotrack_window_support PUBLIC rk3576_yolo_host OpenSSL::Crypto)
target_include_directories(nanotrack_window_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(nanotrack_window_recovery main.cpp)
include(${CMAKE_CURRENT_SOURCE_DIR}/../baseline/code_identity.cmake)
baseline_code_identity(nanotrack_window_recovery "${REPOSITORY_ROOT}")
target_include_directories(nanotrack_window_recovery PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(nanotrack_window_recovery PRIVATE nanotrack_window_support rk3576_yolo_sdk nanotrack_baseline_support ${CMAKE_DL_LIBS})
